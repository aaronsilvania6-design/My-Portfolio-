let score = 0;

function moveSignal() {
  const x = Math.random();
  const y = Math.random();
  signal.style.left = x * 85 + "%";
  signal.style.top = y * 70 + "%";
}

signal.addEventListener("click", () => {
  score++;
  updateScore(score);
  moveSignal();
});
